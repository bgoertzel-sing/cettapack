HE 0.2.10 (PyPI hyperon, user site-packages) aborts on this KB family.

Reproduce:
  python3 bench/gen_kb.py --snps 600 --genes 150 --interactions 750 --annotations 500 --queries 40 --out /tmp/kb_600.metta --queries-out /tmp/q_600.txt
  # load /tmp/kb_600.metta into &self, then: !(match &self (target rs329 $g) $g)
  python3 - /tmp/kb_600.metta /tmp/q_600.txt < he_find.py (script below)

--- he_find.py ---
import sys
from hyperon import MeTTa
m = MeTTa()
s = m.space()
for a in m.parse_all(open(sys.argv[1]).read()):
    s.add_atom(a)
print("loaded", flush=True)
for rs in open(sys.argv[2]).read().split():
    print("q", rs, flush=True)
    r = m.run(f"!(match &self (target {rs} $g) $g)")
    print("  single", r, flush=True)
    r = m.run(f"!(match &self (, (target {rs} $g) (expressed $g $t)) ($g $t))")
    print("  conj", len(r[0]), flush=True)
